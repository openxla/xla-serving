# Python dependencies for examples/dlrm_benchmark.

# (requires Python 3.12–3.14; see docs/installation.md):
#   pip install xla-saved-model xla-serving-data
#   pip install -r examples/dlrm_benchmark/requirements.txt
#

#   pip install "jax[tpu]" -f https://storage.googleapis.com/jax-releases/libtpu_releases.html
#   pip install "jax[cuda12]"   # NVIDIA GPU (or jax[cuda13])
#   pip install jax             # CPU only (smoke tests)
#

# identical JAX / XLA versions.
